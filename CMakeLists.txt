cmake_minimum_required(VERSION 3.20)
project(kalender2026 VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(ENG_USE_LUAJIT OFF CACHE BOOL "Use the portable Lua runtime")
set(ENG_BUILD_TESTS OFF CACHE BOOL "Build upstream tests separately")
set(ENG_BUILD_RUNNER OFF CACHE BOOL "Use the calendar host")
add_subdirectory(engine)

set(WS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/bq_websocket")
set(ws_platform "${WS_DIR}/bq_websocket_platform.c")
if(NOT EMSCRIPTEN)
    # Send SNI as well as verifying the certificate on virtual-hosted TLS services.
    file(READ "${ws_platform}" ws_source)
    string(REPLACE "X509_VERIFY_PARAM_set1_host(param, host, 0);"
        "if (!SSL_set_tlsext_host_name(tls->ssl, host) || !X509_VERIFY_PARAM_set1_host(param, host, 0)) return false;"
        ws_source "${ws_source}")
    set(ws_platform "${CMAKE_CURRENT_BINARY_DIR}/bq_websocket_platform.c")
    file(CONFIGURE OUTPUT "${ws_platform}" CONTENT "${ws_source}" @ONLY)
endif()
add_library(hall_ws STATIC "${WS_DIR}/bq_websocket.c" "${ws_platform}")
target_include_directories(hall_ws PUBLIC "${WS_DIR}")
if(NOT EMSCRIPTEN)
    find_package(OpenSSL REQUIRED)
    find_package(Threads REQUIRED)
    target_compile_definitions(hall_ws PRIVATE BQWS_PT_USE_OPENSSL=1)
    target_link_libraries(hall_ws PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
    if(WIN32)
        target_link_libraries(hall_ws PRIVATE ws2_32)
    endif()
endif()
add_executable(lantern src/entry.c src/main.c src/content.c src/client.c src/settings.c src/feedback_audio.c)
if(NOT EMSCRIPTEN)
    target_sources(lantern PRIVATE server/room.c src/feedback_hud.c)
    target_include_directories(lantern PRIVATE engine/third_party/nuklear)
endif()
target_link_libraries(lantern PRIVATE engine::core hall_ws)
if(UNIX AND NOT APPLE AND NOT EMSCRIPTEN)
    set_target_properties(lantern PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
target_include_directories(lantern PRIVATE src)
option(HALL_COVERAGE "Instrument first-party calendar C" OFF)
if(HALL_COVERAGE AND NOT EMSCRIPTEN)
    target_compile_options(lantern PRIVATE --coverage -O0 -g)
    target_link_options(lantern PRIVATE --coverage)
endif()
if(EMSCRIPTEN)
    target_link_options(lantern PRIVATE "-lidbfs.js" "-sFORCE_FILESYSTEM=1" "-sALLOW_MEMORY_GROWTH=1"
        "-sSTACK_SIZE=2097152" "-sINITIAL_MEMORY=67108864" "-sENVIRONMENT=web"
        "-sEXPORTED_FUNCTIONS=['_main','_malloc','_free']"
        "-sEXPORTED_RUNTIME_METHODS=['callMain','FS','IDBFS','ccall','cwrap','HEAPF32','HEAP32','GL']")
endif()
enable_testing()
if(NOT EMSCRIPTEN)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(hall_units tests/host_unit.c src/main.c src/content.c src/settings.c src/feedback_audio.c src/feedback_hud.c server/room.c)
    target_include_directories(hall_units PRIVATE src engine/third_party/nuklear)
    target_link_libraries(hall_units PRIVATE engine::core hall_ws)
    if(HALL_COVERAGE)
        target_compile_options(hall_units PRIVATE --coverage -O0 -g)
        target_link_options(hall_units PRIVATE --coverage)
    endif()
    add_test(NAME hall_units COMMAND hall_units "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/unit-profile")
    add_test(NAME hall_game COMMAND lantern --root "${CMAKE_CURRENT_SOURCE_DIR}" --headless --doctor --test tests/hall.lua)
    add_test(NAME gallery_route COMMAND lantern --root "${CMAKE_CURRENT_SOURCE_DIR}" --headless --test tests/route.lua)
    add_test(NAME room_network COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/network_test.py" "$<TARGET_FILE:lantern>" "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
